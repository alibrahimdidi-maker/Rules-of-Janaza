فهرس الموضوعين السادس والسابع

6-7 ވަނަ ބައިގައި ބަލައިލާނީ، ކަށުނަމާދުގެ މާތްކަމާއި، ކުރާނެ ފުރިހަމަ ގޮތާއި، ރުކުންތަކާއި (ނިޔަތްގަތުން، 4 ތަކްބީރު، ކުޅަދާނަވެއްޖެނަމަ ކޮޅަށްހުރެ ކުރުން) ފުރިހަމަ ދުޢާތަކާއި ޞަލަވާތުގެ ސީޣާތަކުގެ މައްޗަށެވެ.

فائدة الصلاة للميت

ފުޤަހާއިން ބަޔާންކުރައްވާ ގޮތުގައި، ކަށުނަމާދަކީ، މައްޔިތާއަށްޓަކައި ދިރިހުރި ފަރާތްތަކުން ދަންނަވާ ޝަފާޢަތްތެރި ދުޢާއެކެވެ. ބައިވެރިވި މީހުން ގިނަވި ވަރަކަށް، ފުޤަހާ ބަޔާންކުރައްވާ ގޮތުގައި، ފައިދާ ބޮޑުވެގެންދެއެވެ (ބައްލަވާ ސްލައިޑް 15).

«لقول الله تعالى: ﴿وَمَا أُمِرُوا إِلَّا لِيَعْبُدُوا اللَّهَ مُخْلِصِينَ لَهُ الدِّينَ﴾، وقول رسول الله ﷺ: إِنَّمَا الْأَعْمَالُ بِالنِّيَّاتِ».

الركن الثاني: القيام مع القدرة

ފުޤަހާއިން ބަޔާންކުރައްވާ ގޮތުގައި، ކަށުނަމާދަކީ ފަރުޟު ނަމާދެއް ފަދައިން، ކުޅަދާނަވެއްޖެނަމަ ކޮޅަށްހުރެ ކުރުން ފަރުޟުވެގެންވެއެވެ — ރުކޫޢާއި ސުޖޫދު ނެތުމަކީ، ކޮޅަށްހުރުމުގެ ފަރުޟު ދޫކޮށްލެވޭ ސަބަބެއް ނޫނެވެ. ބަލިވެ ކޮޅަށްނުހުރެވޭ މީހަކަށް، އިށީނދެގެން ކުރެވިދާނެއެވެ.

الركن الرابع: التكبيرة الثانية — الصلاة (الصيغ الكاملة)

«وتؤدَّى الصلاة والسلام على رسول الله ﷺ بأي صيغة، ولو قال: اللهم صل على محمد لكفى، لكن اتباع المأثور أفضل» (ص.364).

① ކުރު ސީޣާ:

«اللَّهُمَّ صَلِّ عَلَى مُحَمَّدٍ»

② ފުރިހަމަ إبراهيمية ސީޣާ:

«اللَّهُمَّ صَلِّ عَلَى مُحَمَّدٍ وَعَلَى آلِ مُحَمَّدٍ، كَمَا صَلَّيْتَ عَلَى إِبْرَاهِيمَ وَعَلَى آلِ إِبْرَاهِيمَ، إِنَّكَ حَمِيدٌ مَجِيدٌ. اللَّهُمَّ بَارِكْ عَلَى مُحَمَّدٍ وَعَلَى آلِ مُحَمَّدٍ، كَمَا بَارَكْتَ عَلَى إِبْرَاهِيمَ وَعَلَى آلِ إِبْرَاهِيمَ، فِي الْعَالَمِينَ إِنَّكَ حَمِيدٌ مَجِيدٌ»

(މާނަ: އޭ ﷲ، محمد އަށާއި محمد ގެ آل އަށް رحمة ލައްވާނދޭވެ، إبراهيم އަށާއި إبراهيم ގެ آل އަށް رحمة ލެއްވި ފަދައިން)

الركن الخامس: التكبيرة الثالثة — الدعاء

«إِذَا صَلَّيْتُمْ عَلَى الْمَيِّتِ فَأَخْلِصُوا لَهُ الدُّعَاءَ» (رواه أبو داود وابن ماجه).

① ކުރު ދުޢާ:

«اللَّهُمَّ اغْفِرْ لَهُ وَارْحَمْهُ»

② މެދު ދުޢާ (رواه مسلم عن عوف بن مالك):

«اللَّهُمَّ اغْفِرْ لَهُ وَارْحَمْهُ، وَعَافِهِ وَاعْفُ عَنْهُ، وَأَكْرِمْ نُزُلَهُ، وَوَسِّعْ مَدْخَلَهُ، وَاغْسِلْهُ بِمَاءٍ وَثَلْجٍ وَبَرَدٍ»

③ ދިގު ދުޢާ (ފުރިހަމަ، رواه مسلم):

«...وَنَقِّهِ مِنَ الْخَطَايَا كَمَا يُنَقَّى الثَّوْبُ الْأَبْيَضُ مِنَ الدَّنَسِ، وَأَبْدِلْهُ دَارًا خَيْرًا مِنْ دَارِهِ، وَأَهْلًا خَيْرًا مِنْ أَهْلِهِ، وَزَوْجًا خَيْرًا مِنْ زَوْجِهِ، وَأَدْخِلْهُ الْجَنَّةَ، وَأَعِذْهُ مِنْ عَذَابِ الْقَبْرِ وَعَذَابِ النَّارِ»

الركن السادس: التكبيرة الرابعة — السلام

سُنة ވެގެންވަނީ، 1 ސަލާމެއް ކަނާތް ފަރާތަށް ދިނުމެވެ. ސަލާމުގެ ކުރިން ބައެއް ބޭފުޅުން ދުޢާއެއްވެސް ސުންނަތްކުރައްވައެވެ: «اللَّهُمَّ لَا تَحْرِمْنَا أَجْرَهُ، وَلَا تَفْتِنَّا بَعْدَهُ» (رواه أحمد، ص.366).

الدعاء على الطفل (الصيغ الكاملة)
